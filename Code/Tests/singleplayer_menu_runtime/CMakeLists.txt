cmake_minimum_required(VERSION 3.16)
project(singleplayer_menu_runtime CXX)
get_filename_component(source "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(ext "${source}/Code/BF2/External")
file(GLOB vm "${ext}/gameswf/gameswf_*.cpp")
list(FILTER vm EXCLUDE REGEX "gameswf_(xmlsocket|render_handler_|sound_handler_|test_|processor|tools|parser|morph$|morph\\.)")
set(base config container file_util image image_filters jpeg membuf tu_file tu_random tu_timer tu_types utf8 utility zlib_adapter)
foreach(name IN LISTS base)
  list(APPEND vm "${ext}/base/${name}.cpp")
endforeach()
add_executable(singleplayer_menu_runtime probe.cpp ${vm})
target_include_directories(singleplayer_menu_runtime PRIVATE "${ext}" "${ext}/gameswf")
set_target_properties(singleplayer_menu_runtime PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES)
target_compile_options(singleplayer_menu_runtime PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/HostCompat.h" -fpermissive -fno-strict-aliasing -fwrapv)
